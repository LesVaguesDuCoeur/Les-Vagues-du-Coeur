function doGet(e) {
  var docId = '1hj6uSP1ygTEK7B6Zf9AN4bwzIVgnqslC1csFj-XyRyA';
  if (!e || !e.parameter || !e.parameter.content) {
    return ContentService.createTextOutput("Erreur: contenu manquant");
  }

  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var doc = DocumentApp.openById(docId);
    var body = doc.getBody();
    body.clear();
    body.appendParagraph(e.parameter.content);
    return ContentService.createTextOutput("Succès");
  } catch(err) {
    return ContentService.createTextOutput("Erreur: " + err.toString());
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    // 1. Get Content from POST body
    var content = "";
    if (e.postData && e.postData.contents) {
       content = e.postData.contents;
    } else if (e.parameter.content) {
       content = e.parameter.content;
    }

    // 2. Determine Action
    var action = e.parameter.action;

    if (action === "log_visit") {
        // --- LOGGING VISITOR ---
        var logDocId = "1v8MDUecYzM4ZKz8ik2TBfTWPt6e_AueiaXMF2LjMOfE";
        var doc = DocumentApp.openById(logDocId);
        var body = doc.getBody();
        // Append log line
        body.appendParagraph(content);
        return ContentService.createTextOutput("Log Success");

    } else if (action === "replace") {
        // --- SYNC RECIPES ---
        var recipeDocId = "1hj6uSP1ygTEK7B6Zf9AN4bwzIVgnqslC1csFj-XyRyA";
        var doc = DocumentApp.openById(recipeDocId);
        var body = doc.getBody();
        // Clear and Replace
        body.clear();
        body.setText(content);
        return ContentService.createTextOutput("Sync Success");
    }

    return ContentService.createTextOutput("Unknown Action");

  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.toString());
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  return ContentService.createTextOutput("Use POST to update.");
}
